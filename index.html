Google Apps Script (`Code.gs`) Code:

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    if (data.tasks && data.tasks.length > 0) {
      data.tasks.forEach(function(task) {
        var imageUrl = "";
        if (task.photo) {
          var folder = DriveApp.getFoldersByName("App_Task_Photos");
          var targetFolder = folder.hasNext() ? folder.next() : DriveApp.createFolder("App_Task_Photos");
          var base64Data = task.photo.split(',')[1];
          var decoded = Utilities.base64Decode(base64Data);
          var blob = Utilities.newBlob(decoded, "image/png", "Task_" + task.id + ".png");
          var file = targetFolder.createFile(blob);
          imageUrl = file.getUrl();
        }

        sheet.appendRow([
          task.id, task.title, task.dateTime, 
          task.priority, task.timePriority, 
          task.notifyMethod, task.message, 
          imageUrl, task.completed ? "Yes" : "No"
        ]);

        if (task.notifyMethod === "Email") {
          MailApp.sendEmail(
            Session.getActiveUser().getEmail(),
            "Task Reminder: " + task.title,
            "Priority: " + task.priority + "\nDue: " + task.dateTime + "\nMessage: " + task.message
          );
        }
      });
    }
    return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT);
  } catch (err) {
    return ContentService.createTextOutput("Error: " + err.toString()).setMimeType(ContentService.MimeType.TEXT);
  }
}
